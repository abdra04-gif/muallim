\ifdefstring{\docstyle}{thesis}{%
  \AtBeginDocument{\ifdefined\chapter

    \titleformat{name=\chapter,numberless}[display]
      {\normalfont\headfamily\bfseries\color{textaccent}}{}{0pt}{\huge}
